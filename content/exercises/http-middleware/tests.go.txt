package main

import (
	"fmt"
	"net/http"
	"net/http/httptest"
	"strings"
)

func serve(h http.Handler, req *http.Request) *httptest.ResponseRecorder {
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, req)
	return rec
}

func tag(name string, trail *[]string) Middleware {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			*trail = append(*trail, name+">")
			next.ServeHTTP(w, r)
			*trail = append(*trail, "<"+name)
		})
	}
}

func main() {
	ok := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { w.WriteHeader(200) })

	runTests([]testCase{
		{"Chain runs the first middleware outermost", func(t *T) {
			var trail []string
			h := Chain(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { trail = append(trail, "handler") }),
				tag("A", &trail), tag("B", &trail))
			serve(h, httptest.NewRequest("GET", "/", nil))
			got, want := strings.Join(trail, " "), "A> B> handler <B <A"
			if got != want {
				t.Errorf("order = %q, want %q", got, want)
			}
		}},
		{"RequestID generates an id when missing", func(t *T) {
			var seen string
			h := RequestID(func() string { return "req-123" })(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				seen = r.Header.Get("X-Request-ID")
			}))
			rec := serve(h, httptest.NewRequest("GET", "/", nil))
			if seen != "req-123" {
				t.Errorf("handler saw X-Request-ID %q, want req-123", seen)
			}
			if got := rec.Header().Get("X-Request-ID"); got != "req-123" {
				t.Errorf("response X-Request-ID = %q, want req-123", got)
			}
		}},
		{"RequestID keeps an incoming id", func(t *T) {
			generated := false
			h := RequestID(func() string { generated = true; return "new" })(ok)
			req := httptest.NewRequest("GET", "/", nil)
			req.Header.Set("X-Request-ID", "from-gateway")
			rec := serve(h, req)
			if generated || rec.Header().Get("X-Request-ID") != "from-gateway" {
				t.Errorf("incoming id was not propagated (response %q, generated=%v)", rec.Header().Get("X-Request-ID"), generated)
			}
		}},
		{"Recover turns a panic into a JSON 500", func(t *T) {
			var logged string
			h := Recover(func(f string, a ...any) { logged = fmt.Sprintf(f, a...) })(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				var m map[string]int
				m["boom"]++ // nil map write panics
			}))
			rec := serve(h, httptest.NewRequest("POST", "/v1/orders", nil))
			if rec.Code != 500 {
				t.Errorf("status = %d, want 500", rec.Code)
			}
			if body := strings.TrimSpace(rec.Body.String()); body != `{"error":"internal"}` {
				t.Errorf("body = %q, want {\"error\":\"internal\"}", body)
			}
			if !strings.Contains(logged, "/v1/orders") {
				t.Errorf("log line %q should mention the path", logged)
			}
		}},
		{"Recover is transparent when nothing panics", func(t *T) {
			h := Recover(func(string, ...any) {})(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				w.WriteHeader(201)
				w.Write([]byte("created"))
			}))
			rec := serve(h, httptest.NewRequest("GET", "/", nil))
			if rec.Code != 201 || rec.Body.String() != "created" {
				t.Errorf("got %d %q, want 201 \"created\"", rec.Code, rec.Body.String())
			}
		}},
		{"full stack: request id survives a panic", func(t *T) {
			h := Chain(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) { panic("bug") }),
				RequestID(func() string { return "trace-me" }), Recover(func(string, ...any) {}))
			rec := serve(h, httptest.NewRequest("GET", "/", nil))
			if rec.Code != 500 || rec.Header().Get("X-Request-ID") != "trace-me" {
				t.Errorf("got status %d, X-Request-ID %q; want 500 with trace-me", rec.Code, rec.Header().Get("X-Request-ID"))
			}
		}},
	})
}
