// testkit.go — the tiny test runner used by every Microverse exercise.
// It is compiled alongside your exercise.go and the exercise's tests.
// Avoid declaring these names in your code: T, testCase, runTests.
package main

import (
	"fmt"
	"os"
	"strings"
	"time"
)

// T mirrors the parts of testing.T the exercises use.
type T struct {
	failed  bool
	stopped bool // set by Fatalf, which ends the test with a panic
	msgs    []string
}

func (t *T) Errorf(format string, args ...any) {
	t.failed = true
	t.msgs = append(t.msgs, fmt.Sprintf(format, args...))
}

func (t *T) Fatalf(format string, args ...any) {
	t.Errorf(format, args...)
	t.stopped = true
	panic("test stopped by Fatalf")
}

type testCase struct {
	name string
	fn   func(t *T)
}

// runTests runs each test with panic recovery and a 3-second deadline, then
// prints machine-readable results the Code Lab turns into a checklist.
func runTests(tests []testCase) {
	passed := 0
	for _, tc := range tests {
		t := &T{}
		done := make(chan any, 1)
		go func() {
			defer func() { done <- recover() }()
			tc.fn(t)
		}()
		select {
		case r := <-done:
			if r != nil && !t.stopped {
				t.Errorf("panic: %v", r)
			}
		case <-time.After(3 * time.Second):
			t.Errorf("timed out after 3s — a deadlock, or a goroutine that never returns?")
		}
		if t.failed {
			fmt.Printf("FAIL  %s :: %s\n", tc.name, strings.Join(t.msgs, " | "))
		} else {
			passed++
			fmt.Printf("PASS  %s\n", tc.name)
		}
	}
	fmt.Printf("RESULT %d/%d\n", passed, len(tests))
	if passed != len(tests) {
		os.Exit(1)
	}
}
