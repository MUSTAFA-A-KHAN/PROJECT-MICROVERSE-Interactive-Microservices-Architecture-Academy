package main

import (
	"sync"
	"sync/atomic"
	"time"
)

type clock struct {
	mu sync.Mutex
	t  time.Time
}

func (c *clock) now() time.Time { c.mu.Lock(); defer c.mu.Unlock(); return c.t }
func (c *clock) advance(d time.Duration) { c.mu.Lock(); c.t = c.t.Add(d); c.mu.Unlock() }

func spend(b *Bucket, n int) (allowed int) {
	for i := 0; i < n; i++ {
		if b.Allow() {
			allowed++
		}
	}
	return
}

func main() {
	runTests([]testCase{
		{"a new bucket allows a burst of Capacity", func(t *T) {
			c := &clock{t: time.Unix(0, 0)}
			b := NewBucket(5, 1, c.now)
			if got := spend(b, 8); got != 5 {
				t.Errorf("allowed %d of 8 immediate requests, want 5", got)
			}
		}},
		{"refills at Rate tokens per second", func(t *T) {
			c := &clock{t: time.Unix(0, 0)}
			b := NewBucket(5, 2, c.now)
			spend(b, 5)
			c.advance(1500 * time.Millisecond) // earns 3 tokens
			if got := spend(b, 5); got != 3 {
				t.Errorf("after 1.5s at 2/s allowed %d, want 3", got)
			}
		}},
		{"never refills above Capacity", func(t *T) {
			c := &clock{t: time.Unix(0, 0)}
			b := NewBucket(3, 10, c.now)
			c.advance(time.Hour)
			if got := spend(b, 10); got != 3 {
				t.Errorf("after an idle hour allowed %d, want 3 (Capacity)", got)
			}
		}},
		{"RetryAfter says when the next token arrives", func(t *T) {
			c := &clock{t: time.Unix(0, 0)}
			b := NewBucket(1, 4, c.now) // one token every 250ms
			if d := b.RetryAfter(); d != 0 {
				t.Errorf("full bucket RetryAfter = %v, want 0", d)
			}
			b.Allow()
			c.advance(100 * time.Millisecond)
			if d := b.RetryAfter(); d < 149*time.Millisecond || d > 151*time.Millisecond {
				t.Errorf("RetryAfter = %v, want ~150ms", d)
			}
		}},
		{"Limiter keeps an independent bucket per key", func(t *T) {
			c := &clock{t: time.Unix(0, 0)}
			l := &Limiter{Capacity: 2, Rate: 1, Now: c.now}
			a := 0
			for i := 0; i < 4; i++ {
				if l.Allow("alice") {
					a++
				}
			}
			if a != 2 {
				t.Errorf("alice allowed %d of 4, want 2", a)
			}
			if !l.Allow("bob") {
				t.Errorf("bob was limited by alice's traffic")
			}
		}},
		{"exactly Capacity allowed under concurrent load", func(t *T) {
			c := &clock{t: time.Unix(0, 0)}
			l := &Limiter{Capacity: 100, Rate: 1, Now: c.now}
			var ok atomic.Int64
			var wg sync.WaitGroup
			for i := 0; i < 20; i++ {
				wg.Add(1)
				go func() {
					defer wg.Done()
					for j := 0; j < 20; j++ {
						if l.Allow("shared") {
							ok.Add(1)
						}
					}
				}()
			}
			wg.Wait()
			if ok.Load() != 100 {
				t.Errorf("allowed %d of 400 concurrent requests, want exactly 100", ok.Load())
			}
		}},
	})
}
