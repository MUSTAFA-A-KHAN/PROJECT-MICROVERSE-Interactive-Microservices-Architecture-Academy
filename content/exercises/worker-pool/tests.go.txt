package main

import (
	"context"
	"errors"
	"fmt"
	"sync/atomic"
	"time"
)

// tracked returns fn that squares its input after a delay and records peak concurrency.
func tracked(delay time.Duration, peak *atomic.Int32) func(context.Context, int) (int, error) {
	var current atomic.Int32
	return func(ctx context.Context, n int) (int, error) {
		c := current.Add(1)
		defer current.Add(-1)
		for {
			p := peak.Load()
			if c <= p || peak.CompareAndSwap(p, c) {
				break
			}
		}
		select {
		case <-time.After(delay):
			return n * n, nil
		case <-ctx.Done():
			return 0, ctx.Err()
		}
	}
}

func main() {
	jobs := []int{1, 2, 3, 4, 5, 6, 7, 8}
	runTests([]testCase{
		{"results are correct and in input order", func(t *T) {
			var peak atomic.Int32
			out, err := Process(context.Background(), jobs, 3, tracked(5*time.Millisecond, &peak))
			if err != nil {
				t.Fatalf("unexpected error: %v", err)
			}
			if fmt.Sprint(out) != "[1 4 9 16 25 36 49 64]" {
				t.Errorf("got %v, want [1 4 9 16 25 36 49 64]", out)
			}
		}},
		{"runs concurrently, never above the limit", func(t *T) {
			var peak atomic.Int32
			start := time.Now()
			_, err := Process(context.Background(), jobs, 4, tracked(40*time.Millisecond, &peak))
			if err != nil {
				t.Fatalf("unexpected error: %v", err)
			}
			if p := peak.Load(); p < 2 || p > 4 {
				t.Errorf("peak concurrency = %d, want between 2 and 4 (workers=4)", p)
			}
			if el := time.Since(start); el > 250*time.Millisecond {
				t.Errorf("took %v; 8 jobs × 40ms on 4 workers should take ~80ms", el)
			}
		}},
		{"one worker means one at a time", func(t *T) {
			var peak atomic.Int32
			_, _ = Process(context.Background(), []int{1, 2, 3}, 1, tracked(5*time.Millisecond, &peak))
			if p := peak.Load(); p != 1 {
				t.Errorf("peak concurrency = %d with workers=1", p)
			}
		}},
		{"first error is returned and the rest are cancelled", func(t *T) {
			boom := errors.New("job 3 failed")
			var cancelled atomic.Int32
			start := time.Now()
			_, err := Process(context.Background(), jobs, 4, func(ctx context.Context, n int) (int, error) {
				if n == 3 {
					return 0, boom
				}
				select {
				case <-time.After(time.Second):
					return n, nil
				case <-ctx.Done():
					cancelled.Add(1)
					return 0, ctx.Err()
				}
			})
			if !errors.Is(err, boom) {
				t.Errorf("err = %v, want the job-3 error", err)
			}
			if el := time.Since(start); el > 500*time.Millisecond {
				t.Errorf("took %v; other calls should be cancelled, not waited for", el)
			}
			if cancelled.Load() == 0 {
				t.Errorf("no in-flight call observed ctx cancellation")
			}
		}},
		{"empty input", func(t *T) {
			var peak atomic.Int32
			out, err := Process(context.Background(), nil, 3, tracked(time.Millisecond, &peak))
			if err != nil || len(out) != 0 {
				t.Errorf("got (%v, %v), want ([], nil)", out, err)
			}
		}},
	})
}
